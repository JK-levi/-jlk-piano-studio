seq.push({type:"cover", num:1, section:"FRONT MATTER"});
seq.push({type:"howto", num:2, section:"FRONT MATTER"});
EASY_PAGES.forEach((d,i)=>seq.push({type:"chords", num:EASY_START+i, section:"EASY CHORDS", data:d}));
TECHNIQUE_PAGES.forEach((d,i)=>seq.push({type:"technique", num:TECH_START+i, section:"PLAYING TECHNIQUES", data:d}));
MORE_PROGRESSIONS.forEach((d,i)=>seq.push({type:"chords", num:MORE_START+i, section:"MORE PROGRESSIONS", data:d}));
MOOD_PAGES.forEach((d,i)=>seq.push({type:"chords", num:MOOD_START+i, section:"MOOD LOOPS", data:d}));
seq.push({type:"divider", num:SONG_DIVIDER, section:"FAVORITE SONG VIBES"});
SONG_PAGES.forEach((d,i)=>seq.push({type:"chords", num:SONG_START+i, section:"FAVORITE SONG VIBES", data:d}));
seq.push({type:"keepplaying", num:KEEP_PLAYING_NUM, section:"BACK MATTER"});

const sections = [
  {name:"Easy Chords", items:EASY_PAGES.map((d,i)=>({title:d.title, num:EASY_START+i}))},
  {name:"Techniques", items:TECHNIQUE_PAGES.map((d,i)=>({title:d.title, num:TECH_START+i}))},
  {name:"More Progressions", items:MORE_PROGRESSIONS.map((d,i)=>({title:d.title, num:MORE_START+i}))},
  {name:"Mood Loops", items:MOOD_PAGES.map((d,i)=>({title:d.title, num:MOOD_START+i}))},
  {name:"Song Vibes", items:SONG_PAGES.map((d,i)=>({title:d.title, num:SONG_START+i}))},
];

// =====================================================================
// RENDER
// =====================================================================
const contentEl = document.getElementById('content');
const stageEl = document.getElementById('stage');
const pageLabelEl = document.getElementById('pageLabel');
const dotsEl = document.getElementById('dots');
let idx = 0;

function esc(s){ return String(s).replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;"); }

function linkifyPageRefs(text){
  // turn "page 8", "p. 9", "page 17)" etc. into tappable jump links
  return esc(text).replace(/\b(page|p\.)\s?(\d{1,2})\b/gi, (m, word, n) => {
    return `<span class="link" data-jump="${n}">${word} ${n}</span>`;
  });
}

function renderChordPage(d){
  let html = '';
  if(d.easy) html += `<div class="badge">EASY</div>`;
  html += `<div class="title">${esc(d.title)}</div>`;
  html += `<div class="subtitle">${esc(d.mood)}</div>`;
  html += `<div class="box"><div class="l"><div class="k">GO:KEYS 3</div><div class="v">${esc(d.patch)}</div></div>`;
  html += `<div class="r"><div class="k">PLAYING NOTE</div><div class="v">${linkifyPageRefs(d.note)}</div></div></div>`;
  d.loops.forEach(([name, chords])=>{
    html += `<div class="loop"><div class="loopname">${linkifyPageRefs(name)}</div>`;
    chords.forEach(([chord,rh,lh])=>{
      html += `<div class="row"><div class="chord">${esc(chord)}</div><div class="lh">LH: ${esc(lh)}</div><div class="rh">RH: ${esc(rh)}</div></div>`;
    });
    html += `</div>`;
  });
  html += `<div class="footnote">${linkifyPageRefs(d.footer)}</div>`;
  return html;
}

function renderTechniquePage(d){
  let html = `<div class="title">${esc(d.title)}</div>`;
  html += `<div class="subtitle">${esc(d.subtitle)}</div>`;
  html += `<ul class="bullets">` + d.bullets.map(b=>`<li>${esc(b)}</li>`).join('') + `</ul>`;
  html += `<hr class="hr">`;
  html += `<div class="tryit-label">TRY IT</div>`;
  html += `<div class="tryit-intro">${linkifyPageRefs(d.tryIntro.replace('{page}', d.pageRef))}</div>`;
  html += `<ul class="steps">` + d.trySteps.map(s=>`<li>${esc(s)}</li>`).join('') + `</ul>`;
  return html;
}

function renderDivider(){
  return `<div class="center-screen">
    <div class="cover-h1">FAVORITE<br>SONG VIBES</div>
    <div class="cover-p" style="margin-top:14px;">Familiar harmonic worlds reduced to something you can sit down and play.</div>
    <div class="cover-p2">These are simplified chord/vibe sketches rather than full transcriptions.</div>
  </div>`;
}

function renderCover(){
  return `<div class="center-screen">
    <div class="cover-h1">KEYBOARD<br>PROGRESSION BOOK</div>
    <div class="cover-sub">SECOND EDITION · POCKET VIEW</div>
    <div class="cover-p">Sit down. Pick a page. Start playing.<br>Every page is self-contained — RH notes and LH bass side by side.</div>
    <div class="cover-stat">3 EASY + 6 TECHNIQUES + 4 MORE PROGRESSIONS<br>+ 20 MOOD LOOPS + 11 SONG VIBES</div>
  </div>`;
}

function renderHowTo(){
  return `<div class="center-screen">
    <div class="title" style="font-size:19px;">HOW TO USE IT</div>
    <div class="cover-p" style="margin-top:14px;">No required order. On every chord line, LH (left hand / bass note) sits right next to RH (right hand notes, played together unless you arpeggiate them). Change pedal with the harmony unless a page says otherwise.</div>
    <div class="cover-p2">Tap <b style="color:var(--accent)">Contents</b> above to jump to any section, or swipe / use Prev·Next to move through the whole book in order.</div>
  </div>`;
}

function renderKeepPlaying(){
  return `<div class="center-screen">
    <div class="title" style="font-size:19px;">KEEP PLAYING</div>
    <div class="cover-p">If one loop grabs you, stay there. Change the rhythm. Move the bass an octave. Break the RH apart. Or do absolutely nothing to it and just enjoy the sound.</div>
    <div class="cover-p2">GO:KEYS 3 tone names in this booklet are drawn from Roland's GO:KEYS 3 / GO:KEYS 5 Tone List.</div>
  </div>`;
}

function fitToScreen(){
  contentEl.style.transform = 'translateX(-50%) scale(1)';
  const avail = stageEl.clientHeight;
  const needed = contentEl.scrollHeight;
  const scale = needed > avail ? Math.max(0.55, avail / needed) : 1;
  contentEl.style.transform = `translateX(-50%) scale(${scale})`;
}

function render(){
  const item = seq[idx];
  let html = '';
  if(item.type === 'chords') html = renderChordPage(item.data);
  else if(item.type === 'technique') html = renderTechniquePage(item.data);
  else if(item.type === 'divider') html = renderDivider();
  else if(item.type === 'cover') html = renderCover();
  else if(item.type === 'howto') html = renderHowTo();
  else if(item.type === 'keepplaying') html = renderKeepPlaying();
  contentEl.innerHTML = html;

  pageLabelEl.textContent = `${item.section} · p.${item.num}`;
  dotsEl.textContent = `${idx+1} / ${seq.length}`;
  document.getElementById('btnPrev').disabled = idx === 0;
  document.getElementById('btnNext').disabled = idx === seq.length - 1;

  requestAnimationFrame(fitToScreen);
  try{ localStorage.setItem('kpb-idx', String(idx)); }catch(e){}
}

function goTo(i){
  idx = Math.max(0, Math.min(seq.length - 1, i));
  render();
}
function goToPageNum(n){
  const i = seq.findIndex(s => s.num === n);
  if(i >= 0) goTo(i);
}

document.getElementById('btnNext').addEventListener('click', ()=>goTo(idx+1));
document.getElementById('btnPrev').addEventListener('click', ()=>goTo(idx-1));
document.getElementById('btnHome').addEventListener('click', ()=>goTo(0));

contentEl.addEventListener('click', (e)=>{
  const j = e.target.closest('[data-jump]');
  if(j) goToPageNum(parseInt(j.dataset.jump, 10));
});

// swipe
let touchX = null, touchY = null;
stageEl.addEventListener('touchstart', (e)=>{
  touchX = e.touches[0].clientX; touchY = e.touches[0].clientY;
}, {passive:true});
stageEl.addEventListener('touchend', (e)=>{
  if(touchX === null) return;
  const dx = e.changedTouches[0].clientX - touchX;
  const dy = e.changedTouches[0].clientY - touchY;
  if(Math.abs(dx) > 50 && Math.abs(dx) > Math.abs(dy)*1.5){
    if(dx < 0) goTo(idx+1); else goTo(idx-1);
  }
  touchX = null; touchY = null;
}, {passive:true});

// prevent iOS bounce/zoom gestures
document.addEventListener('gesturestart', e=>e.preventDefault());
document.addEventListener('touchmove', e=>{ if(e.scale && e.scale !== 1) e.preventDefault(); }, {passive:false});

window.addEventListener('resize', ()=>requestAnimationFrame(fitToScreen));
window.addEventListener('orientationchange', ()=>setTimeout(fitToScreen, 200));

// MENU / CONTENTS

const menuEl = document.getElementById('menu');
const tabsEl = document.getElementById('tabs');
const menuListEl = document.getElementById('menuList');
let activeTab = 0;

function renderTabs(){
  tabsEl.innerHTML = sections.map((s,i)=>
    `<button data-tab="${i}" class="${i===activeTab?'active':''}">${esc(s.name)}</button>`
  ).join('');
}
function renderMenuList(){
  const s = sections[activeTab];
  menuListEl.innerHTML = s.items.map(it=>
    `<div class="menu-item" data-num="${it.num}"><div class="t">${esc(it.title)}</div><div class="p">${it.num}</div></div>`
  ).join('');
}
function openMenu(){
  activeTab = 0;
  renderTabs(); renderMenuList();
  menuEl.classList.add('open');
}
function closeMenu(){ menuEl.classList.remove('open'); }

document.getElementById('btnMenu').addEventListener('click', openMenu);
document.getElementById('btnCloseMenu').addEventListener('click', closeMenu);
tabsEl.addEventListener('click', (e)=>{
  const b = e.target.closest('button[data-tab]');
  if(!b) return;
  activeTab = parseInt(b.dataset.tab, 10);
  renderTabs(); renderMenuList();
});
menuListEl.addEventListener('click', (e)=>{
  const item = e.target.closest('.menu-item');
  if(!item) return;
  goToPageNum(parseInt(item.dataset.num, 10));
  closeMenu();
});

(function init(){
  let start = 0;
  try{
    const saved = parseInt(localStorage.getItem('kpb-idx'), 10);
    if(!isNaN(saved) && saved >= 0 && saved < seq.length) start = saved;
  }catch(e){}
  goTo(start);
})();
</script>
